2. Script IoT para Envio Remoto via Python / ESP32 (REST API Firebase)

import requests
import datetime

# Firebase REST API Endpoint
PROJECT_ID = "SEU-FIREBASE-PROJECT-ID"
URL = f"https://firestore.googleapis.com/v1/projects/{PROJECT_ID}/databases/(default)/documents/artifacts/decibelimetro-pro/public/data/acoustic_logs"

payload = {
    "fields": {
        "environment": {"stringValue": "Sensor Automático ESP32"},
        "tag": {"stringValue": "IOT-01"},
        "zone": {"stringValue": "Setor Producao"},
        "db": {"doubleValue": 87.5},
        "riskLevel": {"stringValue": "Perigoso"},
        "timestamp": {"stringValue": datetime.datetime.utcnow().isoformat() + "Z"}
    }
}

response = requests.post(URL, json=payload)
print("Log Enviado:", response.status_code)
